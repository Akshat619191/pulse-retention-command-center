QUERY LAB

From signal to action

⚡ 0.84 sec
WITH customer_signal AS (
  SELECT customer_id, MAX(order_date) last_order,
         SUM(net_revenue) AS ltv
  FROM fact_orders
  WHERE order_date >= DATEADD(day, -90, CURRENT_DATE)
  GROUP BY 1
)
SELECT * FROM customer_signal
WHERE ltv > 5000 AND last_order < DATEADD(day, -21, CURRENT_DATE);
